CREATE DATABASE PROYECTO;
GO
USE PROYECTO;
GO

CREATE TABLE METODO_PAGO
(
  ID_Metodo INT IDENTITY(1,1) NOT NULL, 
  Descripcion VARCHAR(50) NOT NULL,
  CONSTRAINT PK_MP_ID_Metodo PRIMARY KEY (ID_Metodo)
);

CREATE TABLE CLIENTE
(
  DNI INT NOT NULL, -- NO lleva IDENTITY (es un dato real)
  Nombre VARCHAR(50) NOT NULL,
  Apellido VARCHAR(50) NOT NULL,
  Telefono VARCHAR(20) NOT NULL,
  Calle VARCHAR(50) NOT NULL,
  Altura INT NOT NULL,
  Localidad VARCHAR(50) NOT NULL,
  CONSTRAINT PK_CLIENTE_DNI PRIMARY KEY (DNI)
);

CREATE TABLE VENTA
(
  Nro_Comprobante INT IDENTITY(1,1) NOT NULL, 
  Fecha DATETIME NOT NULL,
  DNI INT NOT NULL,
  CONSTRAINT PK_V_Nro_Comprobante PRIMARY KEY (Nro_Comprobante),
  CONSTRAINT FK_VENTA_DNI FOREIGN KEY (DNI) REFERENCES CLIENTE(DNI)
    ON DELETE NO ACTION
    ON UPDATE CASCADE, 
  CONSTRAINT CHK_VENTA_Fecha CHECK (Fecha <= GETDATE())
);

CREATE TABLE CATEGORIA
(
  ID_Categoria INT IDENTITY(1,1) NOT NULL, 
  Nombre VARCHAR(50) NOT NULL UNIQUE, 
  CONSTRAINT PK_C_ID_Categoria PRIMARY KEY (ID_Categoria)
);

CREATE TABLE PRENDA
(
  Codigo_SKU INT NOT NULL, 
  Talle VARCHAR(10) NOT NULL, 
  Color VARCHAR(50) NOT NULL,
  Precio_Actual DECIMAL(10,2) NOT NULL,
  Stock INT NOT NULL CONSTRAINT CHK_PRENDA_Stock CHECK (Stock >= 0), 
  Modelo VARCHAR(50) NOT NULL,
  ID_Categoria INT NOT NULL,
  CONSTRAINT PK_P_Codigo_SKU PRIMARY KEY (Codigo_SKU),
  CONSTRAINT FK_PRENDA_ID_Categoria FOREIGN KEY (ID_Categoria) REFERENCES CATEGORIA(ID_Categoria)
    ON DELETE NO ACTION
    ON UPDATE CASCADE, 
  CONSTRAINT CHK_PRENDA_Precio_Actual CHECK (Precio_Actual > 0)
);

CREATE TABLE ABONA_CON
(
  ID_Metodo INT NOT NULL,
  Nro_Comprobante INT NOT NULL,
  Monto_Abonado DECIMAL(10,2) NOT NULL,
  CONSTRAINT PK_ABONA_CON PRIMARY KEY (ID_Metodo, Nro_Comprobante),
  CONSTRAINT FK_ABONA_CON_ID_Metodo FOREIGN KEY (ID_Metodo) REFERENCES METODO_PAGO(ID_Metodo)
    ON DELETE NO ACTION
    ON UPDATE CASCADE,
  CONSTRAINT FK_ABONA_CON_Nro_Comprobante FOREIGN KEY (Nro_Comprobante) REFERENCES VENTA(Nro_Comprobante)
    ON DELETE CASCADE
    ON UPDATE CASCADE, -- Faltaba una coma aquí en tu script original
  CONSTRAINT CHK_ABONA_CON_Monto_Abonado CHECK (Monto_Abonado > 0)
);

CREATE TABLE CONTIENE
(
  Nro_Comprobante INT NOT NULL,
  Codigo_SKU INT NOT NULL,
  Precio_Historico DECIMAL(10,2) NOT NULL,
  Cantidad INT NOT NULL,
  CONSTRAINT PK_CONTIENE PRIMARY KEY (Nro_Comprobante, Codigo_SKU),
  CONSTRAINT FK_CONTIENE_Nro_Comprobante FOREIGN KEY (Nro_Comprobante) REFERENCES VENTA(Nro_Comprobante)
    ON DELETE CASCADE
    ON UPDATE CASCADE,
  CONSTRAINT FK_CONTIENE_Codigo_SKU FOREIGN KEY (Codigo_SKU) REFERENCES PRENDA(Codigo_SKU)
    ON DELETE NO ACTION
    ON UPDATE CASCADE, -- Faltaba una coma aquí en tu script original
  CONSTRAINT CHK_CONTIENE_Precio_Historico CHECK (Precio_Historico > 0),
  CONSTRAINT CHK_CONTIENE_Cantidad CHECK (Cantidad > 0)
);
